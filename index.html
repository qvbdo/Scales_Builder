📋 Google Apps Script Setup Guide

Step 1: Create the Script

  1. Open your Google Sheet
  2. Click Extensions → Apps Script
  3. Delete the default myFunction
  4. Paste the code from the Data Hub below

Step 2: Deploy as Web App

  1. Click Deploy → New Deployment
  2. Select type: Web App
  3. Execute as: Me
  4. Who has access: Anyone
  5. Click Deploy and authorize
  6. Copy the Web App URL

Step 3: Connect to Platform

  1. Paste the URL in Builder → "Google Sheet URL"
  2. Click Test Connection to verify
  3. Export Online App and start collecting!

// === GOOGLE APPS SCRIPT (Code.gs) ===
// Deploy as Web App: Execute as Me, Access: Anyone

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    
    if (data.test === true) {
      return ContentService.createTextOutput(JSON.stringify({
        success: true, 
        message: "Connection test successful",
        timestamp: new Date().toISOString()
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Submissions");
    if (!sheet) {
      sheet = ss.insertSheet("Submissions");
    }
    
    var baseHeaders = ["id","participantId","projectId","timestamp","duration_sec","syncStatus","checksum"];
    var answerKeys = [];
    if (data.answers) {
      Object.keys(data.answers).forEach(function(k) {
        if (k !== "_duration_sec" && !k.startsWith("_")) answerKeys.push(k);
      });
    }
    var allHeaders = baseHeaders.concat(answerKeys);
    
    var existingHeaders = [];
    if (sheet.getLastRow() > 0) {
      existingHeaders = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    }
    
    if (existingHeaders.length === 0 || existingHeaders[0] === "" || existingHeaders[0] === null) {
      sheet.getRange(1, 1, 1, allHeaders.length).setValues([allHeaders]);
      sheet.getRange(1, 1, 1, allHeaders.length)
        .setFontWeight("bold").setBackground("#4285f4").setFontColor("white");
      existingHeaders = allHeaders;
    } else {
      var newHeaders = [];
      allHeaders.forEach(function(h) {
        if (existingHeaders.indexOf(h) === -1) newHeaders.push(h);
      });
      if (newHeaders.length > 0) {
        var startCol = existingHeaders.length + 1;
        sheet.getRange(1, startCol, 1, newHeaders.length).setValues([newHeaders]);
        sheet.getRange(1, startCol, 1, newHeaders.length)
          .setFontWeight("bold").setBackground("#4285f4").setFontColor("white");
        existingHeaders = existingHeaders.concat(newHeaders);
      }
    }
    
    var meta = {
      id: data.id || "",
      participantId: data.participantId || "",
      projectId: data.projectId || "",
      timestamp: data.timestamp ? new Date(data.timestamp).toISOString() : "",
      duration_sec: data.answers && data.answers._duration_sec ? data.answers._duration_sec : "",
      syncStatus: data._syncStatus || "synced",
      checksum: data._checksum || ""
    };
    
    var row = [];
    existingHeaders.forEach(function(h) {
      if (meta.hasOwnProperty(h)) row.push(meta[h]);
      else if (data.answers && data.answers.hasOwnProperty(h)) {
        var val = data.answers[h];
        if (Array.isArray(val)) row.push(val.join("|"));
        else if (typeof val === "string" && val.length > 50000) row.push(val.substring(0, 50000) + "...[truncated]");
        else row.push(val);
      } else row.push("");
    });
    
    sheet.appendRow(row);
    
    return ContentService.createTextOutput(JSON.stringify({
      success: true, 
      message: "Record saved", 
      id: data.id, 
      row: sheet.getLastRow()
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      success: false, 
      error: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({
    status: "active", 
    version: "1.0", 
    timestamp: new Date().toISOString()
  })).setMimeType(ContentService.MimeType.JSON);
}
